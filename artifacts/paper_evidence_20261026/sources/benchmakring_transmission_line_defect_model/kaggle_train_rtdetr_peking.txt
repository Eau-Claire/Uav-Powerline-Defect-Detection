# Cell 3: Automatic YOLO-to-COCO JSON Annotation Generator (Runs in ~5 seconds!)
from PIL import Image
from tqdm import tqdm

CLASS_NAMES = [
    "insulator",
    "insulator_broken",
    "insulator_flashover",
    "bird_nest",
    "foreign_object",
    "broken_strand"
]

ann_dir = dest_dataset_dir / "annotations"
ann_dir.mkdir(parents=True, exist_ok=True)

def convert_yolo_to_coco(split_name: str):
    out_json = ann_dir / f"{split_name}.json"
    if out_json.exists():
        print(f"✅ Đã có sẵn {out_json.name}. Bỏ qua bước tạo lại.")
        return out_json
    
    print(f"Đang tạo COCO JSON cho split '{split_name}'...")
    img_dir = dest_dataset_dir / "images" / split_name
    lbl_dir = dest_dataset_dir / "labels" / split_name
    
    images_list = []
    annotations_list = []
    categories = [{"id": i, "name": name} for i, name in enumerate(CLASS_NAMES)]
    
    ann_id = 0
    img_files = sorted(list(img_dir.glob("*.jpg")) + list(img_dir.glob("*.jpeg")) + list(img_dir.glob("*.png")))
    
    for img_id, img_path in enumerate(tqdm(img_files, desc=f"Converting {split_name}")):
        with Image.open(img_path) as im:
            w, h = im.size
            
        images_list.append({
            "id": img_id,
            "file_name": img_path.name,
            "width": w,
            "height": h
        })
        
        lbl_path = lbl_dir / f"{img_path.stem}.txt"
        if lbl_path.exists():
            with open(lbl_path, "r", encoding="utf-8") as f:
                for line in f:
                    parts = line.strip().split()
                    if len(parts) >= 5:
                        cls_id = int(parts[0])
                        xc, yc, bw, bh = [float(x) for x in parts[1:5]]
                        # Convert normalized [xc, yc, w, h] to COCO [xmin, ymin, w, h]
                        xmin = (xc - bw / 2.0) * w
                        ymin = (yc - bh / 2.0) * h
                        box_w = bw * w
                        box_h = bh * h
                        area = box_w * box_h
                        
                        annotations_list.append({
                            "id": ann_id,
                            "image_id": img_id,
                            "category_id": cls_id,
                            "bbox": [round(xmin, 2), round(ymin, 2), round(box_w, 2), round(box_h, 2)],
                            "area": round(area, 2),
                            "iscrowd": 0
                        })
                        ann_id += 1
                        
    coco_dict = {
        "images": images_list,
        "annotations": annotations_list,
        "categories": categories
    }
    with open(out_json, "w", encoding="utf-8") as f:
        json.dump(coco_dict, f)
    print(f"✅ Tạo xong {out_json.name}: {len(images_list):,} ảnh, {len(annotations_list):,} nhãn!")
    return out_json

train_json = convert_yolo_to_coco("train")
val_json = convert_yolo_to_coco("val")

# Bulletproof Image Processor:
# RT-DETR official config (PekingU/rtdetr_r50vd) specifies:
# resize to 640x640 (Bilinear), rescale by 1/255, do_normalize=False.
class PowerlineImageProcessor:
    def __init__(self, size=(640, 640)):
        self.size = size
        
    @classmethod
    def from_pretrained(cls, *args, **kwargs):
        return cls(size=(640, 640))
        
    def save_pretrained(self, save_dir):
        import json
        from pathlib import Path
        config = {
            "do_normalize": False,
            "do_rescale": True,
            "do_resize": True,
            "image_processor_type": "RTDetrImageProcessor",
            "rescale_factor": 1.0 / 255.0,
            "size": {"height": self.size[0], "width": self.size[1]}
        }
        p = Path(save_dir) / "preprocessor_config.json"
        p.parent.mkdir(parents=True, exist_ok=True)
        with open(p, "w", encoding="utf-8") as f:
            json.dump(config, f, indent=2)
            
    def __call__(self, images, return_tensors="pt"):
        if not isinstance(images, list):
            images = [images]
        tensors = []
        for img in images:
            img_rgb = img.convert("RGB").resize(self.size, Image.BILINEAR)
            arr = np.array(img_rgb, dtype=np.float32).transpose(2, 0, 1) / 255.0
            tensors.append(torch.from_numpy(arr))
        return {"pixel_values": torch.stack(tensors) if return_tensors == "pt" else tensors}

class PowerlineCOCODataset(Dataset):
    def __init__(self, img_dir, json_path, processor):
        self.img_dir = Path(img_dir)
        self.processor = processor
        with open(json_path, "r", encoding="utf-8") as f:
            coco = json.load(f)
            
        self.images = {img["id"]: img for img in coco["images"]}
        self.img_ids = list(self.images.keys())
        
        # Group annotations by image_id
        self.img_to_anns = {i: [] for i in self.img_ids}
        for ann in coco["annotations"]:
            self.img_to_anns[ann["image_id"]].append(ann)
            
    def __len__(self):
        return len(self.img_ids)
        
    def __getitem__(self, idx):
        img_id = self.img_ids[idx]
        img_info = self.images[img_id]
        img_path = self.img_dir / img_info["file_name"]
        
        image = Image.open(img_path).convert("RGB")
        w, h = img_info["width"], img_info["height"]
        
        anns = self.img_to_anns[img_id]
        boxes = []
        classes = []
        for ann in anns:
            # COCO bbox: [xmin, ymin, bw, bh]
            xmin, ymin, bw, bh = ann["bbox"]
            # Convert to normalized [xc, yc, w, h] in [0, 1] as required by RT-DETR
            xc = (xmin + bw / 2.0) / w
            yc = (ymin + bh / 2.0) / h
            nw = bw / w
            nh = bh / h
            boxes.append([xc, yc, nw, nh])
            classes.append(ann["category_id"])
            
        if not boxes:
            boxes = torch.zeros((0, 4), dtype=torch.float32)
            classes = torch.zeros((0,), dtype=torch.long)
        else:
            boxes = torch.tensor(boxes, dtype=torch.float32)
            classes = torch.tensor(classes, dtype=torch.long)
            
        # Process image
        inputs = self.processor(images=image, return_tensors="pt")
        pixel_values = inputs["pixel_values"].squeeze(0)
        
        target = {"class_labels": classes, "boxes": boxes}
        return pixel_values, target

def collate_fn(batch):
    pixel_values = torch.stack([item[0] for item in batch])
    targets = [item[1] for item in batch]
    return pixel_values, targets

    """
    assert model_variant in ["r50vd", "r101vd"], "model_variant must be 'r50vd' or 'r101vd'"
    repo_id = f"PekingU/rtdetr_{model_variant}"
    run_name = f"raw_rtdetr_{model_variant}"
    save_dir = Path(f"/kaggle/working/runs/{run_name}")
    save_dir.mkdir(parents=True, exist_ok=True)
    
    print("=" * 70)
    print(f"LAUNCHING TRAINING: {run_name.upper()}")
    print(f"  Base HuggingFace Model : {repo_id}")
    print(f"  Epochs                 : {epochs}")
    print(f"  Batch Size             : {batch_size}")
    print(f"  Optimizer              : AdamW (backbone: 1e-4, head: 1e-3, cos_lr)")
    print("=" * 70)
    
    # 1. Load Processor & Datasets
    processor = RTDetrImageProcessor.from_pretrained(repo_id)
    train_ds = PowerlineCOCODataset(dest_dataset_dir / "images" / "train", train_json, processor)
    val_ds = PowerlineCOCODataset(dest_dataset_dir / "images" / "val", val_json, processor)
    
    train_loader = DataLoader(train_ds, batch_size=batch_size, shuffle=True, num_workers=4, collate_fn=collate_fn, pin_memory=True)
    val_loader = DataLoader(val_ds, batch_size=batch_size, shuffle=False, num_workers=4, collate_fn=collate_fn, pin_memory=True)
    
    # 2. Load Model with official pre-trained weights
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"Loading {repo_id} weights from Hugging Face...")
    try:
        from transformers import RTDetrConfig
        config = RTDetrConfig.from_pretrained(repo_id)
        config.num_labels = len(CLASS_NAMES)
        config.id2label = id2label
        config.label2id = label2id
        model = RTDetrForObjectDetection.from_pretrained(
            repo_id,
            config=config,
            ignore_mismatched_sizes=True
        )
    except Exception:
        model = RTDetrForObjectDetection.from_pretrained(
            repo_id,
            num_labels=len(CLASS_NAMES),
            id2label=id2label,
            label2id=label2id,
            ignore_mismatched_sizes=True
        )
    
    # Direct Single-GPU Execution (avoids DataParallel List[Dict] target mismatch)
    model = model.to(device)
    gpu_name = torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'CPU'
    print(f"✅ Model moved to {device} ({gpu_name})")
    
    # 3. Optimizer & Cosine Scheduler as required by Benchmark Proposal
    # Differential learning rate: 1e-4 for backbone, 1e-3 for transformer
    backbone_params = []
    other_params = []
    for name, param in model.named_parameters():
        if not param.requires_grad:
            continue
        if "backbone" in name:
            backbone_params.append(param)
        else:
            other_params.append(param)
            
    optimizer = optim.AdamW([
        {"params": backbone_params, "lr": 1e-4},
        {"params": other_params, "lr": 1e-3}
    ], weight_decay=0.0005)
    
    scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs, eta_min=1e-5)
    scaler = torch.cuda.amp.GradScaler()
    
    # Check for resume
    last_ckpt_path = save_dir / "last_checkpoint.pt"
    best_ckpt_path = save_dir / "best_model.pt"
    start_epoch = 0
    best_val_loss = float("inf")
    
    if last_ckpt_path.exists():
        print(f"[RESUME] Found existing checkpoint: {last_ckpt_path}. Resuming...")
        ckpt = torch.load(last_ckpt_path, map_location=device)
        (model.module if hasattr(model, 'module') else model).load_state_dict(ckpt["model_state"])
        optimizer.load_state_dict(ckpt["optimizer_state"])
        scheduler.load_state_dict(ckpt["scheduler_state"])
        start_epoch = ckpt["epoch"] + 1
        best_val_loss = ckpt.get("best_val_loss", float("inf"))
        print(f"Resumed successfully from Epoch {start_epoch}!")
        
    # 4. Training Loop
    print(f"\n>>> Starting Training Loop ({start_epoch+1} to {epochs}) <<<")
    for epoch in range(start_epoch, epochs):
        model.train()
        train_loss = 0.0
        pbar = tqdm(train_loader, desc=f"Epoch {epoch+1}/{epochs} [Train]")
        
        optimizer.zero_grad()
        for step_idx, (pixel_values, targets) in enumerate(pbar):
            pixel_values = pixel_values.to(device)
            gpu_targets = [{k: v.to(device) for k, v in t.items()} for t in targets]
            
            with torch.cuda.amp.autocast():
                outputs = model(pixel_values=pixel_values, labels=gpu_targets)
                loss = outputs.loss.mean() if hasattr(outputs.loss, 'mean') else outputs.loss
                loss = loss / accum_steps
                
            scaler.scale(loss).backward()
            
            if (step_idx + 1) % accum_steps == 0 or (step_idx + 1) == len(train_loader):
                scaler.unscale_(optimizer)
                torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=0.1)
                scaler.step(optimizer)
                scaler.update()
                optimizer.zero_grad()
            
            train_loss += loss.item() * accum_steps
            pbar.set_postfix({"loss": f"{loss.item() * accum_steps:.4f}"})
            
        scheduler.step()
        avg_train_loss = train_loss / len(train_loader)
        
        # Validation
        model.eval()
        val_loss = 0.0
        with torch.no_grad():
            for pixel_values, targets in tqdm(val_loader, desc=f"Epoch {epoch+1}/{epochs} [Val]", leave=False):
                pixel_values = pixel_values.to(device)
                gpu_targets = [{k: v.to(device) for k, v in t.items()} for t in targets]
                with torch.cuda.amp.autocast():
                    outputs = model(pixel_values=pixel_values, labels=gpu_targets)
                    loss = outputs.loss.mean() if hasattr(outputs.loss, 'mean') else outputs.loss
                    val_loss += loss.item()
                    
        avg_val_loss = val_loss / len(val_loader)
        print(f"Epoch {epoch+1:02d} Summary -> Train Loss: {avg_train_loss:.4f} | Val Loss: {avg_val_loss:.4f} | LR: {scheduler.get_last_lr()[0]:.6f}")
        
        # Save Checkpoint (every epoch for last, best for minimum val loss)
        raw_model = model.module if hasattr(model, 'module') else model
        save_payload = {
            "epoch": epoch,
            "model_state": raw_model.state_dict(),
            "optimizer_state": optimizer.state_dict(),
            "scheduler_state": scheduler.state_dict(),
            "best_val_loss": best_val_loss,
            "train_loss": avg_train_loss,
            "val_loss": avg_val_loss
        }
        torch.save(save_payload, last_ckpt_path)
        
        if avg_val_loss < best_val_loss:
            best_val_loss = avg_val_loss
            torch.save(save_payload, best_ckpt_path)
            raw_model.save_pretrained(save_dir / "best_huggingface_model")
            processor.save_pretrained(save_dir / "best_huggingface_model")
            print(f"  🌟 New best model saved! (Val Loss: {best_val_loss:.4f})")
            
    print(f"\n[DONE] Finished training for {run_name.upper()}! Saved in: {save_dir}")
    
    # Auto-sync model output to Camber Cloud Stash if enabled
    if os.environ.get("CAMBER_API_KEY") and "myusername" not in globals().get("CAMBER_STASH_DEST", "myusername"):
        try:
            dest = globals().get("CAMBER_STASH_DEST", "").rstrip("/") + f"/{run_name}/"
            print(f"☁️ Đang tự động backup {run_name} lên Camber Stash: {dest}...")
            import subprocess
            res = subprocess.run(["camber", "stash", "cp", "-r", str(save_dir), dest], capture_output=True, text=True)
            if res.returncode == 0:
                print(f"✅ Đã backup an toàn {run_name} lên Camber Cloud Stash!")
            else:
                print(f"⚠️ Camber upload notice: {res.stderr[:200]}")
        except Exception as e:
            print(f"⚠️ Camber sync warning: {e}")
    
    # Clean up GPU VRAM so the next model starts with 100% clean memory
    del model, optimizer, scheduler, train_loader, val_loader
    import gc; gc.collect()
    torch.cuda.empty_cache()
    print("🧹 Đã dọn dẹp bộ nhớ VRAM sạch sẽ để sẵn sàng cho model tiếp theo.")
    return best_ckpt_path
