    break

    raise FileNotFoundError("Could not find 'images/train' in /kaggle/input/. Please add your raw_combined dataset to this notebook!")

dest_dataset_dir.mkdir(parents=True, exist_ok=True)

# Scan all source files
print("Scanning dataset files...")
all_src_files = [p for p in src_dataset_dir.rglob("*.*") if p.is_file() and not p.name.endswith(".cache")]
total_files = len(all_src_files)

# Filter files that need to be copied (skip already copied!)
files_to_copy = []
for src_file in all_src_files:
    rel_path = src_file.relative_to(src_dataset_dir)
    dst_file = dest_dataset_dir / rel_path
    if not dst_file.exists():
        files_to_copy.append((src_file, dst_file))

already_copied = total_files - len(files_to_copy)
print(f"📊 Trạng thái hiện tại: Đã có sẵn {already_copied:,} / {total_files:,} files ({already_copied/total_files*100:.1f}%) trong NVMe SSD (/tmp)")

if files_to_copy:
    print(f"🚀 Đang copy {len(files_to_copy):,} file còn lại bằng 16 luồng song song...")
    def _copy_worker(pair):
        s, d = pair
        d.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(s, d)
        
    with ThreadPoolExecutor(max_workers=16) as pool:
        for _ in tqdm(pool.map(_copy_worker, files_to_copy), total=len(files_to_copy), desc="Copying to SSD", unit="file"):
            pass
    print("✅ Hoàn tất copy 100% dữ liệu sang /tmp!")

# Cell 3: Standardized RT-DETR Training Function
from ultralytics import RTDETR

def train_rtdetr(model_name: str = "r50"):
    """
    Trains RT-DETR-R50 or RT-DETR-R101 strictly following the Benchmark Proposal.
    Supports automatic multi-GPU training (2x T4 DDP) and resuming if interrupted.
    """
    assert model_name in ["r50", "r101"], "model_name must be 'r50' or 'r101'"
    
    weights_file = "rtdetr-l.pt" if model_name == "r50" else "rtdetr-x.pt"
    run_name = f"raw_rtdetr_{model_name}"
    run_dir = Path(f"/kaggle/working/runs/{run_name}")
    run_dir.mkdir(parents=True, exist_ok=True)
    
    last_pt = run_dir / "weights" / "last.pt"
    is_resuming = last_pt.exists()
    
    # Multi-GPU support: if 2 GPUs detected, use device=[0, 1] (PyTorch DDP)
    num_gpus = torch.cuda.device_count()
    device_arg = [0, 1] if num_gpus >= 2 else 0
    
    # Batch size: effective_batch = 16 across runs as specified in proposal
    batch_size = 16 if model_name == "r50" else 8
    
    print("=" * 70)
    print(f"STARTING BENCHMARK RUN: {run_name.upper()}")
    print(f"  Architecture : RT-DETR-{model_name.upper()} ({weights_file})")
    print(f"  Mode         : {'RESUMING from ' + str(last_pt) if is_resuming else 'FRESH START'}")
    print(f"  Device       : {device_arg} ({num_gpus} GPUs)")
    print(f"  Batch Size   : {batch_size}")
    print(f"  Image Size   : 640")
    print(f"  Epochs       : 50")
    print(f"  Optimizer    : AdamW (lr0=0.001, cos_lr=True)")
    print("=" * 70)
    
    if is_resuming:
        model = RTDETR(str(last_pt))
    else:
        model = RTDETR(weights_file)
        
    results = model.train(
        data=str(data_yaml_path),
        epochs=50,
        imgsz=640,
        batch=batch_size,
        device=device_arg,
        optimizer="AdamW",
        lr0=0.001,
        lrf=0.01,
        cos_lr=True,
        warmup_epochs=3.0,
        weight_decay=0.0005,
        close_mosaic=10,
        amp=True,
        workers=4,
        project="/kaggle/working/runs",
        name=run_name,
        exist_ok=True,
        save=True,
        save_period=5,
        plots=True,
        resume=is_resuming
    )
    
    print(f"\n[DONE] Finished {run_name.upper()} successfully!")
    return results

# Estimated time: ~3.5 - 4.5 hours on 2x T4 (~7h on 1x T4)
